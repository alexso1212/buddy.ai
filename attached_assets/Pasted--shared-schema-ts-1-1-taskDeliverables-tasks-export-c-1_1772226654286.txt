=== 1.1 新建 taskDeliverables 表 ===

在 tasks 表定义之后，新增：

export const taskDeliverables = pgTable("task_deliverables", {
  id: serial("id").primaryKey(),
  taskId: integer("task_id").notNull().references(() => tasks.id, { onDelete: "cascade" }),
  orgId: integer("org_id").notNull().references(() => organizations.id),
  
  // 交付物类型
  type: varchar("type", { length: 50 }).notNull(),  // "file" | "link" | "text" | "ai_generated"
  
  // 内容（根据 type 使用不同字段）
  title: text("title").notNull(),                     // 交付物标题，如 "产品调研报告 v2"
  description: text("description"),                    // 补充说明
  
  // file 类型
  fileUrl: text("file_url"),                          // 文件存储地址
  fileName: text("file_name"),                        // 原始文件名
  fileSize: integer("file_size"),                     // 文件大小（字节）
  fileMimeType: varchar("file_mime_type", { length: 200 }),  // MIME 类型
  
  // link 类型
  linkUrl: text("link_url"),                          // 外部链接
  
  // text 类型
  content: text("content"),                           // 文本内容（支持 Markdown）
  
  // 提交信息
  submittedBy: integer("submitted_by").notNull().references(() => users.id),
  submittedAt: timestamp("submitted_at").defaultNow().notNull(),
  
  // 版本控制
  version: integer("version").default(1).notNull(),   // 版本号，重新提交时递增
  isLatest: boolean("is_latest").default(true).notNull(),  // 是否为最新版本
  
  // 审核结果（P4/P5 阶段会用到，先预留字段）
  reviewStatus: varchar("review_status", { length: 50 }).default("pending"),  // pending | approved | rejected | scored
  reviewScore: integer("review_score"),               // AI 评分 0-100
  reviewFeedback: text("review_feedback"),            // AI 审核反馈
  reviewedBy: integer("reviewed_by").references(() => users.id),
  reviewedAt: timestamp("reviewed_at"),
  
  createdAt: timestamp("created_at").defaultNow().notNull(),
  updatedAt: timestamp("updated_at").defaultNow().notNull(),
});

=== 1.3 新增任务提交记录表 ===

export const taskSubmissions = pgTable("task_submissions", {
  id: serial("id").primaryKey(),
  taskId: integer("task_id").notNull().references(() => tasks.id, { onDelete: "cascade" }),
  orgId: integer("org_id").notNull().references(() => organizations.id),
  submittedBy: integer("submitted_by").notNull().references(() => users.id),
  
  // 提交说明
  note: text("note"),                                  // 提交时的备注，如 "第二版，修改了数据部分"
  
  // 关联的交付物 ID 列表（JSON 数组）
  deliverableIds: jsonb("deliverable_ids").$type<number[]>().default([]),
  
  // 状态
  status: varchar("status", { length: 50 }).default("pending").notNull(),  // pending | approved | rejected | revision_requested
  
  // 审核
  reviewedBy: integer("reviewed_by").references(() => users.id),
  reviewedAt: timestamp("reviewed_at"),
  reviewNote: text("review_note"),                    // 审核意见
  overallScore: integer("overall_score"),             // 综合评分 0-100
  
  createdAt: timestamp("created_at").defaultNow().notNull(),
});

=== 1.5 添加 relations ===

在 relations 部分添加：

export const taskDeliverablesRelations = relations(taskDeliverables, ({ one }) => ({
  task: one(tasks, { fields: [taskDeliverables.taskId], references: [tasks.id] }),
  organization: one(organizations, { fields: [taskDeliverables.orgId], references: [organizations.id] }),
  submitter: one(users, { fields: [taskDeliverables.submittedBy], references: [users.id] }),
  reviewer: one(users, { fields: [taskDeliverables.reviewedBy], references: [users.id] }),
}));

export const taskSubmissionsRelations = relations(taskSubmissions, ({ one }) => ({

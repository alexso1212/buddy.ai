# task-18: 管理后台监控面板

> **预计耗时**：20-30 分钟（页面多但逻辑不复杂，主要是SQL聚合+前端展示）
> **新增文件**：`server/routes/admin.ts`（管理API集中管理）、`client/src/pages/admin/` 目录（5个页面）
> **改动文件**：`shared/schema.ts`（users表加 isSuperAdmin）、`server/routes.ts`（挂载admin路由）、`App.tsx`（注册admin路由）
> **风险等级**：低（纯新增，不改现有业务逻辑）

-----

## 架构决定：同一项目，路由隔离

```
同一个 Express 服务器
├── /api/*              ← 现有业务 API（authMiddleware）
├── /api/admin/*        ← 管理后台 API（superAdminMiddleware）
└── /admin/*            ← 管理后台前端页面

同一个数据库
├── 业务表（users, tasks, token_usage...）
└── 不新建表，全部用聚合查询读现有数据
```

**为什么不分项目**：

- 共享数据库，不需要跨服务 API 通信
- 共享认证，superAdmin 就是 users 表里的一个标记
- 数据量小（几百用户级别），不会因为管理查询拖慢业务
- 部署和维护只有一套

## STEP 1：数据库改动

### users 表新增 isSuperAdmin 字段

```typescript
isSuperAdmin: boolean('is_super_admin').notNull().default(false),
```

推送后手动标记自己为超级管理员：

```sql
UPDATE users SET is_super_admin = true WHERE email = '你的邮箱';
```

或者用环境变量方式（不改数据库）：

```typescript
// 超管判断逻辑（两种方式取其一）
function isSuperAdmin(user: any): boolean {
  // 方式A：数据库字段
  if (user.isSuperAdmin) return true;
  // 方式B：环境变量白名单
  const adminEmails = (process.env.ADMIN_EMAILS || '').split(',').map(e => e.trim());
  if (adminEmails.includes(user.email)) return true;
  return false;
}
```

**推荐两种都支持**，方便灵活控制。

## STEP 2：超管中间件

```typescript
// server/middleware/superAdmin.ts 或直接写在 admin 路由文件中

async function superAdminMiddleware(req: any, res: any, next: any) {
  // 先走普通 authMiddleware 的逻辑获取 user
  // 如果项目中 authMiddleware 把 user 信息挂在 req 上，直接用
  
  if (!req.currentUserId) {
    return res.status(401).json({ error: '未登录' });
  }
  
  const user = await storage.getUserById(req.currentUserId);
  if (!user || !isSuperAdmin(user)) {
    return res.status(403).json({ error: '无权访问管理后台' });
  }
  
  req.isSuperAdmin = true;
  next();
}
```

// 全员 AI 依赖指数
adminRouter.get("/ai-workforce", async (req, res) => {
  try {
    const period = req.query.period || 'month';
    let dateFilter;
    switch (period) {
      case 'week': dateFilter = sql`>= CURRENT_DATE - INTERVAL '7 days'`; break;
      default: dateFilter = sql`>= DATE_TRUNC('month', CURRENT_DATE)`;
    }

    // 每个用户的 AI 使用数据
    const userData = await db.execute(sql`
      SELECT 
        u.id as user_id,
        u.name,
        u.display_name,
        u.email,
        -- AI 对话数据
        (SELECT COUNT(*) FROM token_usage tu WHERE tu.user_id = u.id AND tu.created_at ${dateFilter}) as ai_conversations,
        (SELECT COALESCE(SUM(tu.total_tokens), 0) FROM token_usage tu WHERE tu.user_id = u.id AND tu.created_at ${dateFilter}) as ai_tokens,
        -- 任务数据
        (SELECT COUNT(*) FROM tasks t WHERE t.assignee_id = u.id AND t.status = 'done' AND t.updated_at ${dateFilter}) as tasks_completed,
        -- 交付物数据（需要检查 task_deliverables 表结构）
        (SELECT COUNT(*) FROM task_deliverables td 
         JOIN tasks t ON td.task_id = t.id 
         WHERE t.assignee_id = u.id AND td.created_at ${dateFilter}) as total_deliverables,
        (SELECT COUNT(*) FROM task_deliverables td 
         JOIN tasks t ON td.task_id = t.id 
         WHERE t.assignee_id = u.id AND td.created_at ${dateFilter}
         AND (td.description LIKE '%ai_chat%' OR td.source = 'ai_chat')) as ai_deliverables,
        -- 任务创建方式
        (SELECT COUNT(*) FROM activity_logs al 
         WHERE al.user_id = u.id AND al.action = 'create_task' AND al.source = 'ai' 
         AND al.created_at ${dateFilter}) as tasks_created_by_ai,
        (SELECT COUNT(*) FROM activity_logs al 
         WHERE al.user_id = u.id AND al.action = 'create_task' AND al.source != 'ai' 
         AND al.created_at ${dateFilter}) as tasks_created_manual
      FROM users u
      WHERE EXISTS (SELECT 1 FROM org_memberships om WHERE om.user_id = u.id)
      ORDER BY ai_tokens DESC
    `);

    // 计算 AI 依赖指数
    const results = userData.rows.map((row: any) => {
      const aiConversations = parseInt(row.ai_conversations) || 0;
      const aiTokens = parseInt(row.ai_tokens) || 0;
      const tasksCompleted = parseInt(row.tasks_completed) || 0;
      const totalDeliverables = parseInt(row.total_deliverables) || 0;
      const aiDeliverables = parseInt(row.ai_deliverables) || 0;
      const tasksAI = parseInt(row.tasks_created_by_ai) || 0;
      const tasksManual = parseInt(row.tasks_created_manual) || 0;

      // AI 交付占比 (40%)
      const deliveryScore = totalDeliverables > 0 ? (aiDeliverables / totalDeliverables) * 100 : 0;
      
      // 对话强度 (25%)
      const convRatio = tasksCompleted > 0 ? aiConversations / tasksCompleted : 0;
      const convScore = Math.min(100, convRatio * 10);
      
      // 任务规划 AI 化 (20%)
      const totalCreated = tasksAI + tasksManual;
      const planScore = totalCreated > 0 ? (tasksAI / totalCreated) * 100 : 0;
      
      // Token 强度 (15%)
      const tokenScore = Math.min(100, aiTokens / 5000);
      
      const index = Math.round(
        deliveryScore * 0.40 + convScore * 0.25 + planScore * 0.20 + tokenScore * 0.15
      );

      let riskLevel: string;
      if (index >= 70) riskLevel = 'critical';
      else if (index >= 50) riskLevel = 'high';
      else if (index >= 20) riskLevel = 'medium';
      else riskLevel = 'low';

      return {
        userId: row.user_id,
        name: row.display_name || row.name,
        email: row.email,
        aiDependencyIndex: Math.min(100, index),
        riskLevel,
        aiConversations,
        aiTokens,
        tasksCompleted,
        totalDeliverables,
        aiDeliverables,
        aiDeliveryRatio: totalDeliverables > 0 ? Math.round(aiDeliverables / totalDeliverables * 100) : 0,
        tasksCreatedByAI: tasksAI,
        tasksCreatedManual: tasksManual,
      };
    });

    // 按 AI 指数排序
    results.sort((a: any, b: any) => b.aiDependencyIndex - a.aiDependencyIndex);

    // 团队整体指数
    const validResults = results.filter((r: any) => r.aiConversations > 0 || r.tasksCompleted > 0);
    const teamIndex = validResults.length > 0
      ? Math.round(validResults.reduce((sum: number, r: any) => sum + r.aiDependencyIndex, 0) / validResults.length)
      : 0;

    // 分布统计
    const distribution = {
      low: results.filter((r: any) => r.riskLevel === 'low').length,
      medium: results.filter((r: any) => r.riskLevel === 'medium').length,
      high: results.filter((r: any) => r.riskLevel === 'high').length,
      critical: results.filter((r: any) => r.riskLevel === 'critical').length,
    };

    res.json({ data: { teamIndex, distribution, members: results } });
  } catch (e: any) {
    res.status(500).json({ error: e.message });
  }
});

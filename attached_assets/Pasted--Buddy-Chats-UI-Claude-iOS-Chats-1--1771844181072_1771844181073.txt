## 3.2 相对时间函数

```typescript
function getRelativeTime(timestamp: number): string {
  const now = Date.now();
  const diff = now - timestamp;
  const minutes = Math.floor(diff / 60000);
  const hours = Math.floor(diff / 3600000);
  const days = Math.floor(diff / 86400000);

  if (minutes < 1) return 'just now';
  if (minutes < 60) return `${minutes} minute${minutes > 1 ? 's' : ''} ago`;
  if (hours < 24) return `${hours} hour${hours > 1 ? 's' : ''} ago`;
  if (days < 7) return `${days} day${days > 1 ? 's' : ''} ago`;
  if (days < 30) return `${Math.floor(days / 7)} week${Math.floor(days / 7) > 1 ? 's' : ''} ago`;
  return `${Math.floor(days / 30)} month${Math.floor(days / 30) > 1 ? 's' : ''} ago`;
}
```

# STEP 4：添加底部固定栏

```
┌──────────────────────────────────────┐
│  🔍 Search                    (＋)   │
└──────────────────────────────────────┘
```

```jsx
function ChatsBottomBar({ onSearch, onNewChat }) {
  const [searchQuery, setSearchQuery] = useState('');

  return (
    <div style={{
      position: 'fixed',
      bottom: 0, left: 0, right: 0,
      padding: '10px 16px',
      paddingBottom: 'calc(10px + env(safe-area-inset-bottom))',
      display: 'flex',
      alignItems: 'center',
      gap: 12,
      background: 'var(--bg-primary)',
    }}>
      {/* 搜索框 — 胶囊形 */}
      <div style={{
        flex: 1, height: 40,
        background: 'rgba(255,255,255,0.08)',
        borderRadius: 20,
        padding: '0 14px',
        display: 'flex', alignItems: 'center', gap: 8,
      }}>
        <SearchIcon size={16} color="#7A7874" />
        <input
          value={searchQuery}
          onChange={e => { setSearchQuery(e.target.value); onSearch(e.target.value); }}
          placeholder="Search"
          style={{
            flex: 1, border: 'none', outline: 'none',
            background: 'transparent',
            fontSize: 15, color: '#ECECEC',
          }}
        />
      </div>

      {/* 新建按钮 — 44px 圆形橙色 */}
      <button onClick={onNewChat} style={{
        width: 44, height: 44, borderRadius: '50%',
        background: '#AE5630', border: 'none',
        display: 'flex', alignItems: 'center', justifyContent: 'center',
        flexShrink: 0, cursor: 'pointer',
      }}>
        <PlusIcon size={22} color="#FFFFFF" strokeWidth={2} />
      </button>
    </div>
  );
}
```

列表底部加 padding 防遮挡：`padding-bottom: 80px;`

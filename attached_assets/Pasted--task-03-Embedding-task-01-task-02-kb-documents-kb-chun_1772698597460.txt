## STEP 1：创建 `server/services/kb/embedding.ts`

```typescript
import { sql } from 'drizzle-orm';
import { db } from '../../db'; // 根据项目实际的 db 导入路径调整

/**
 * Embedding 服务
 * 
 * 方案 A：通过 AI 代理调用 Embedding API
 * 方案 B：如果代理不支持，降级为空操作（依赖全文搜索）
 */

// ========== 配置 ==========
// 根据 STEP 0 的测试结果，设置以下变量：
// 如果代理支持 embedding，设 EMBEDDING_AVAILABLE = true 并填写正确的 URL/KEY/MODEL
// 如果不支持，设 EMBEDDING_AVAILABLE = false

const EMBEDDING_AVAILABLE = true; // ← 根据 STEP 0 结果修改
const EMBEDDING_BASE_URL = process.env.AI_BASE_URL || 'https://vip.aipro.love/v1';
const EMBEDDING_API_KEY = process.env.CLAUDE_SIMPLE_API_KEY || '';
const EMBEDDING_MODEL = 'text-embedding-3-small'; // ← 根据 STEP 0 结果修改
const EMBEDDING_DIMENSIONS = 1536;

// 速率限制：每批次最多处理的 chunk 数
const BATCH_SIZE = 20;
// 每批次之间的延迟（毫秒）
const BATCH_DELAY = 1000;

/**
 * 为单条文本生成 embedding 向量
 */
export async function generateEmbedding(text: string): Promise<number[] | null> {
  if (!EMBEDDING_AVAILABLE) return null;

  try {
    const response = await fetch(`${EMBEDDING_BASE_URL}/embeddings`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${EMBEDDING_API_KEY}`,
      },
      body: JSON.stringify({
        model: EMBEDDING_MODEL,
        input: text.slice(0, 8000), // 截断防止超长
      }),
    });

    if (!response.ok) {
      const errText = await response.text();
      console.error(`[KB Embedding] API error ${response.status}: ${errText}`);
      return null;
    }

    const data = await response.json();
    return data?.data?.[0]?.embedding || null;
  } catch (err: any) {
    console.error(`[KB Embedding] Failed:`, err.message);
    return null;
  }
}

/**
 * 为多个 chunk 批量生成 embedding 并写入数据库
 * 包含速率限制和错误处理
 */
export async function generateEmbeddingsForDocument(documentId: number, chunks: { id: number; content: string }[]): Promise<{ success: number; failed: number }> {
  if (!EMBEDDING_AVAILABLE) {
    console.log(`[KB Embedding] Embedding not available, skipping for document ${documentId}`);
    return { success: 0, failed: 0 };
  }

  let success = 0;
  let failed = 0;

  // 分批处理
  for (let i = 0; i < chunks.length; i += BATCH_SIZE) {
    const batch = chunks.slice(i, i + BATCH_SIZE);

    for (const chunk of batch) {
      try {
        const embedding = await generateEmbedding(chunk.content);
        if (embedding && embedding.length === EMBEDDING_DIMENSIONS) {
          // 用原始 SQL 更新 embedding（因为 Drizzle 可能不原生支持 vector 参数绑定）
          const vectorStr = `[${embedding.join(',')}]`;
          await db.execute(
            sql`UPDATE kb_chunks SET embedding = ${vectorStr}::vector WHERE id = ${chunk.id}`
          );
          success++;
        } else {
          failed++;
          console.warn(`[KB Embedding] Invalid embedding for chunk ${chunk.id}`);
        }
      } catch (err: any) {
        failed++;
        console.error(`[KB Embedding] Error for chunk ${chunk.id}:`, err.message);
      }
    }

    // 批次间延迟（避免触发速率限制）
    if (i + BATCH_SIZE < chunks.length) {
      await new Promise(resolve => setTimeout(resolve, BATCH_DELAY));
    }
  }

  console.log(`[KB Embedding] Document ${documentId}: ${success} success, ${failed} failed`);
  return { success, failed };
}

/**
 * 向量相似度检索
 * 返回与查询最相似的 Top-K 个 chunk
 */
export async function vectorSearch(
  orgId: number,
  queryText: string,
  topK: number = 5
): Promise<{ id: number; documentId: number; content: string; similarity: number; metadata: string | null }[]> {
  if (!EMBEDDING_AVAILABLE) return [];

  const queryEmbedding = await generateEmbedding(queryText);
  if (!queryEmbedding) return [];

  const vectorStr = `[${queryEmbedding.join(',')}]`;

  try {
    const results = await db.execute(
      sql`SELECT 
            c.id, 
            c.document_id as "documentId", 
            c.content, 
            c.metadata,
            1 - (c.embedding <=> ${vectorStr}::vector) as similarity
          FROM kb_chunks c
          JOIN kb_documents d ON c.document_id = d.id
          WHERE c.org_id = ${orgId}
            AND d.status = 'ready'
            AND c.embedding IS NOT NULL
          ORDER BY c.embedding <=> ${vectorStr}::vector
          LIMIT ${topK}`
    );
    return (results.rows || results) as any[];
  } catch (err: any) {
    console.error(`[KB Vector Search] Error:`, err.message);
    return [];
  }
}
```

**注意关于 db 的导入**：

项目中数据库连接可能不是从 `../../db` 导出的。检查 `server/storage.ts` 中的 `db` 是怎么获取的：

- 如果 `db` 是在 `storage.ts` 中创建的局部变量，需要把它 export 出来

## STEP 2：创建 `server/services/kb/search.ts`

这是统一的知识库检索接口，混合了向量检索和全文搜索。

```typescript
import { sql } from 'drizzle-orm';
import { storage } from '../../storage';
import { vectorSearch } from './embedding';

export interface KBSearchResult {
  chunkId: number;
  documentId: number;
  documentTitle: string;
  category: string;
  content: string;
  similarity: number; // 0-1，越高越相关
  source: 'vector' | 'fulltext'; // 来源标记
}

/**
 * 知识库混合检索
 * 
 * 策略：
 * 1. 如果 embedding 可用 → 向量检索 Top-5 + 全文搜索 Top-5 → 合并去重 → 返回 Top-5
 * 2. 如果 embedding 不可用 → 仅全文搜索 Top-5
 * 
 * 权限过滤：
 * - visibility='org' → 所有人可见
 * - visibility='department' → 检查 visibleDeptIds 是否包含用户部门
 * - visibility='admin' → 仅 owner/admin 可见
 */
export async function searchKnowledge(options: {
  orgId: number;
  query: string;
  topK?: number;
  userRole?: string;
  userDeptId?: number | null;
}): Promise<KBSearchResult[]> {
  const { orgId, query, topK = 5, userRole = 'member', userDeptId = null } = options;

  if (!query || query.trim().length < 2) return [];

  // 并行执行向量检索和全文搜索
  const [vectorResults, fulltextResults] = await Promise.all([
    vectorSearch(orgId, query, topK),
    fulltextSearch(orgId, query, topK),
  ]);

  // 合并结果
  const resultMap = new Map<number, KBSearchResult>();

  // 向量结果优先
  for (const r of vectorResults) {
    resultMap.set(r.id, {
      chunkId: r.id,
      documentId: r.documentId,
      documentTitle: '', // 后面补充
      category: '',
      content: r.content,
      similarity: r.similarity,
      source: 'vector',
    });
  }

  // 全文搜索补充（不覆盖已有的向量结果）
  for (const r of fulltextResults) {
    if (!resultMap.has(r.chunkId)) {
      resultMap.set(r.chunkId, r);
    }
  }

  // 转为数组，按 similarity 排序
  let results = Array.from(resultMap.values())
    .sort((a, b) => b.similarity - a.similarity)
    .slice(0, topK);

  // 补充文档标题和分类信息
  const docIds = [...new Set(results.map(r => r.documentId))];
  const docInfoMap = new Map<number, { title: string; category: string; visibility: string; visibleDeptIds: string | null }>();

  for (const docId of docIds) {
    const doc = await storage.getKbDocumentById(docId);
    if (doc) {
      docInfoMap.set(docId, {
        title: doc.title,
        category: doc.category,
        visibility: doc.visibility,
        visibleDeptIds: doc.visibleDeptIds,
      });
    }
  }

  // 填充文档信息 + 权限过滤
  results = results
    .map(r => {
      const docInfo = docInfoMap.get(r.documentId);
      if (!docInfo) return null;

      // 权限过滤
      if (docInfo.visibility === 'admin' && !['owner', 'admin'].includes(userRole)) return null;
      if (docInfo.visibility === 'department' && userDeptId) {
        const visibleDepts = docInfo.visibleDeptIds ? JSON.parse(docInfo.visibleDeptIds) : [];
        if (!visibleDepts.includes(userDeptId)) return null;
      }

      return { ...r, documentTitle: docInfo.title, category: docInfo.category };
    })
    .filter(Boolean) as KBSearchResult[];

  return results;
}

/**
 * 全文搜索（PostgreSQL ILIKE 关键词匹配）
 * 
 * 对中文来说，PostgreSQL 的 tsvector 效果有限（因为缺少分词器）。
 * 所以用 ILIKE 关键词匹配作为主要的全文搜索方式。
 * 简单但有效——用户问"年假政策"，就搜包含"年假"或"政策"的 chunk。
 */
async function fulltextSearch(
  orgId: number,
  query: string,
  topK: number = 5
): Promise<KBSearchResult[]> {
  // 提取关键词（至少 2 个字符的词）
  const keywords = extractKeywords(query);
  if (keywords.length === 0) return [];

  try {
    // 构建 ILIKE 条件：任意一个关键词匹配就算命中
    // 同时计算匹配的关键词数量作为相关度分数
    const conditions = keywords.map(kw => `c.content ILIKE '%${escapeSql(kw)}%'`);
    const matchCount = keywords.map(kw => `CASE WHEN c.content ILIKE '%${escapeSql(kw)}%' THEN 1 ELSE 0 END`);

    const queryStr = `
      SELECT 
        c.id as "chunkId",
        c.document_id as "documentId",
        c.content,
        d.title as "documentTitle",
        d.category,
        (${matchCount.join(' + ')})::float / ${keywords.length} as similarity
      FROM kb_chunks c
      JOIN kb_documents d ON c.document_id = d.id
      WHERE c.org_id = ${orgId}
        AND d.status = 'ready'
        AND (${conditions.join(' OR ')})
      ORDER BY (${matchCount.join(' + ')}) DESC, c.chunk_index ASC
      LIMIT ${topK}
    `;

    const results = await storage.executeRaw(sql.raw(queryStr));
    const rows = (results.rows || results) as any[];

    return rows.map((r: any) => ({
      chunkId: r.chunkId,
      documentId: r.documentId,
      documentTitle: r.documentTitle || '',
      category: r.category || '',
      content: r.content,
      similarity: Math.min(r.similarity || 0, 1), // 归一化到 0-1
      source: 'fulltext' as const,
    }));
  } catch (err: any) {
    console.error(`[KB Fulltext] Error:`, err.message);
    return [];
  }
}

/**
 * 从查询文本中提取关键词
 * 中文按 2-4 字的词组提取，英文按空格分割
 */
function extractKeywords(query: string): string[] {
  const keywords: string[] = [];

  // 英文单词（3 个字母以上）
  const englishWords = query.match(/[a-zA-Z]{3,}/g) || [];
  keywords.push(...englishWords);

  // 中文：先按标点和空格分割成短语，再提取 2-4 字的片段
  const chinesePhrases = query.match(/[\u4e00-\u9fff]{2,}/g) || [];
  for (const phrase of chinesePhrases) {
    if (phrase.length <= 4) {
      keywords.push(phrase);
    } else {
      // 对长短语，用滑动窗口提取 2 字和 3 字的子串
      for (let i = 0; i < phrase.length - 1; i++) {
        keywords.push(phrase.slice(i, i + 2)); // 二字词
      }
      for (let i = 0; i < phrase.length - 2; i++) {
        keywords.push(phrase.slice(i, i + 3)); // 三字词
      }
    }
  }

  // 去重，最多保留 10 个关键词
  return [...new Set(keywords)].slice(0, 10);
}

/**
 * 转义 SQL 中的特殊字符（防注入）
 */
function escapeSql(str: string): string {
  return str.replace(/'/g, "''").replace(/%/g, '\\%').replace(/_/g, '\\_');
}
```

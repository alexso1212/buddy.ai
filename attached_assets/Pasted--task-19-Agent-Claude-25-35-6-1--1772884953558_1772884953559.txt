```
  │                                          │
  │  思考中...                    12秒        │
  │  "让我分析一下这个问题..."                │  ← 思考文本实时显示
  │                                          │
  └──────────────────────────────────────────┘
     点击可展开/折叠思考过程
```

ThinkingBlock 改造：

1. 顶部显示渐变光条（替代 spinner）
1. “思考中…” 文字 + 右侧实时计时器（“12秒”）
1. 思考内容默认折叠，点击”查看思考过程”展开
1. 展开后的思考文字用 `color: var(--text-secondary)` 淡化，和正文区分
1. 思考完成后光条消失，显示”思考了 X 秒”，可点击展开

```typescript
// ThinkingBlock 改造后的结构
function ThinkingBlock({ thinking, isThinking, duration }: Props) {
  const [expanded, setExpanded] = useState(false);
  const [elapsed, setElapsed] = useState(0);
  
  // 实时计时
  useEffect(() => {
    if (!isThinking) return;
    const start = Date.now();
    const timer = setInterval(() => setElapsed(Math.floor((Date.now() - start) / 1000)), 1000);
    return () => clearInterval(timer);
  }, [isThinking]);

  return (
    <div className="my-3">
      {/* 光条 */}
      {isThinking && <div className="thinking-bar mb-2" />}
      
      {/* 标题行 */}
      <button onClick={() => setExpanded(!expanded)} className="flex items-center gap-2 text-sm">
        <ChevronRight className={cn("w-3.5 h-3.5 transition-transform", expanded && "rotate-90")} />
        <span className="text-[var(--text-secondary)]">
          {isThinking ? `思考中...` : `思考了 ${duration || elapsed} 秒`}
        </span>
        {isThinking && (
          <span className="text-xs text-[var(--text-secondary)] tabular-nums">{elapsed}s</span>
        )}
      </button>
      
      {/* 思考内容 */}
      {expanded && thinking && (
        <div className="mt-2 pl-5 text-sm text-[var(--text-secondary)] leading-relaxed whitespace-pre-wrap"
             style={{ maxHeight: 300, overflowY: 'auto' }}>
          {thinking}
        </div>
      )}
    </div>
  );
}
```

```typescript
// 自动调整高度
const textareaRef = useRef<HTMLTextAreaElement>(null);

const autoResize = useCallback(() => {
  const ta = textareaRef.current;
  if (!ta) return;
  ta.style.height = 'auto'; // 先重置
  ta.style.height = Math.min(ta.scrollHeight, 200) + 'px'; // 自适应，最高200px
}, []);

useEffect(() => {
  autoResize();
}, [text, autoResize]);

// textarea 属性
<textarea
  ref={textareaRef}
  value={text}
  onChange={(e) => { setText(e.target.value); autoResize(); }}
  onKeyDown={handleKeyDown}
  rows={1}
  style={{
    resize: 'none',
    minHeight: 24,
    maxHeight: 200,
    overflowY: text.split('\n').length > 8 ? 'auto' : 'hidden',
    transition: 'height 100ms ease',
  }}
/>
```

```typescript
// 在 agent.tsx 中
useEffect(() => {
  if (!window.visualViewport) return;
  const onResize = () => {
    const vv = window.visualViewport!;
    const keyboardHeight = window.innerHeight - vv.height;
    document.documentElement.style.setProperty('--keyboard-height', `${keyboardHeight}px`);

}, []);
```

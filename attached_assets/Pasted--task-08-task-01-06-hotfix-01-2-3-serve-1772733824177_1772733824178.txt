## 目标

用户上传一个 ZIP（或多个文件），AI 自动提取：公司名称、部门结构、岗位职责、文件分类。返回结构化 JSON 供前端展示确认，确认后批量写入数据库。

## STEP 3：创建 `server/services/setup/zipProcessor.ts`

```typescript
import AdmZip from 'adm-zip';
import fs from 'fs';
import path from 'path';
import { extractText } from '../kb/extractText';

export interface ExtractedFile {
  fileName: string;       // 原始文件名（不含路径）
  filePath: string;       // 临时文件路径（用于后续存入KB）
  fileType: string;       // pdf / docx / txt / md
  fileSize: number;       // 字节
  content: string;        // 提取的文本内容（截断到前 10000 字符供 AI 分析）
  fullContent: string;    // 完整文本（用于后续切片入知识库）
}

/**
 * 处理 ZIP 文件：解压 → 提取每个文件的文本
 */
export async function processZipFile(zipPath: string): Promise<ExtractedFile[]> {
  const zip = new AdmZip(zipPath);
  const entries = zip.getEntries();
  const files: ExtractedFile[] = [];
  
  // 创建临时解压目录
  const tempDir = path.join('uploads', 'setup_temp_' + Date.now());
  if (!fs.existsSync(tempDir)) {
    fs.mkdirSync(tempDir, { recursive: true });
  }

  const supportedExts = ['pdf', 'docx', 'txt', 'md'];

  for (const entry of entries) {
    // 跳过目录、隐藏文件、Mac 的 __MACOSX
    if (entry.isDirectory) continue;
    const entryName = entry.entryName;
    if (entryName.startsWith('__MACOSX') || entryName.startsWith('.')) continue;
    if (entryName.includes('/.')) continue; // 隐藏文件在子目录中

    const ext = entryName.split('.').pop()?.toLowerCase() || '';
    if (!supportedExts.includes(ext)) continue;

    // 提取文件名（去掉路径）
    const fileName = entryName.split('/').pop() || entryName;
    const tempPath = path.join(tempDir, Date.now() + '_' + fileName);

    try {
      // 解压单个文件到临时目录
      const fileData = entry.getData();
      fs.writeFileSync(tempPath, fileData);

      // 提取文本
      const fullContent = await extractText(tempPath, ext);
      
      if (fullContent && fullContent.trim().length > 20) {
        files.push({
          fileName,
          filePath: tempPath,
          fileType: ext,
          fileSize: entry.header.size,
          content: fullContent.slice(0, 10000), // AI 分析用的截断版
          fullContent,                           // 完整版用于知识库
        });
      } else {
        // 文本太少，清理临时文件
        if (fs.existsSync(tempPath)) fs.unlinkSync(tempPath);
      }
    } catch (err: any) {
      console.warn(`[Setup] Failed to process ${fileName}:`, err.message);
      if (fs.existsSync(tempPath)) fs.unlinkSync(tempPath);
    }
  }

  console.log(`[Setup] Extracted ${files.length} files from ZIP`);
  return files;
}

/**
 * 处理多个单独上传的文件（非ZIP）
 */
export async function processMultipleFiles(filePaths: { originalName: string; tempPath: string }[]): Promise<ExtractedFile[]> {
  const files: ExtractedFile[] = [];
  const supportedExts = ['pdf', 'docx', 'txt', 'md'];

  for (const { originalName, tempPath } of filePaths) {
    const ext = originalName.split('.').pop()?.toLowerCase() || '';
    if (!supportedExts.includes(ext)) continue;

    try {
      const stats = fs.statSync(tempPath);
      const fullContent = await extractText(tempPath, ext);

      if (fullContent && fullContent.trim().length > 20) {
        files.push({
          fileName: originalName,
          filePath: tempPath,
          fileType: ext,
          fileSize: stats.size,
          content: fullContent.slice(0, 10000),
          fullContent,
        });
      }
    } catch (err: any) {
      console.warn(`[Setup] Failed to process ${originalName}:`, err.message);
    }
  }

  return files;
}

/**
 * 清理临时文件
 */
export function cleanupTempFiles(files: ExtractedFile[]) {
  for (const file of files) {
    try {
      if (fs.existsSync(file.filePath)) {
        fs.unlinkSync(file.filePath);
      }
    } catch {}
  }
  // 清理临时目录
  const tempDirs = fs.readdirSync('uploads').filter(d => d.startsWith('setup_temp_'));
  for (const dir of tempDirs) {
    try {
      fs.rmSync(path.join('uploads', dir), { recursive: true, force: true });
    } catch {}
  }
}
```

```typescript
// ========== 阶段1：Haiku 逐文件分析（快+便宜） ==========

async function analyzeFileWithHaiku(fileName: string, content: string): Promise<FileAnalysis> {
  try {
    const response = await aiClient.chat.completions.create({
      model: 'claude-haiku-4-5-20251001',
      max_tokens: 1000,
      temperature: 0.1,
      messages: [
        {
          role: 'system',
          content: `你是一个企业文档分析助手。分析以下企业文档，返回纯 JSON（不要 markdown 代码块）。

返回格式：
{
  "category": "policy/contract/jd/manual/general 五选一",
  "visibility": "org/admin/department 三选一",
  "visibleDepartment": "如果visibility=department，填部门名（否则填空字符串）",
  "summary": "一句话概括文档内容（30字以内）",
  "mentionedDepartments": ["文档中提到的部门名称列表"],
  "mentionedRoles": ["文档中提到的岗位/职位名称列表"]
}

分类标准：
- policy：规章制度、考勤、休假、行为准则、报销流程
- contract：劳动合同、保密协议、竞业限制
- jd：岗位说明书、职位描述、KPI考核标准
- manual：操作手册、培训资料、使用指南
- general：其他

可见性判断：
- org：全员应知的（考勤、办公规范、报销流程）
- admin：涉及薪资、合同条款、人事的敏感文件
- department：只和特定部门相关的文件`
        },
        {
          role: 'user',
          content: `文件名：${fileName}\n\n文件内容：\n${content.slice(0, 6000)}`
        }
      ],
    });

    const text = response.choices[0]?.message?.content || '';
    const cleaned = text.replace(/```json\s*/g, '').replace(/```\s*/g, '').trim();
    const parsed = JSON.parse(cleaned);

    return {
      fileName,
      category: parsed.category || 'general',
      visibility: parsed.visibility || 'org',
      visibleDepartment: parsed.visibleDepartment || '',
      summary: parsed.summary || '',
      mentionedDepartments: parsed.mentionedDepartments || [],
      mentionedRoles: parsed.mentionedRoles || [],
    };
  } catch (err: any) {
    console.error(`[Setup AI] Haiku analysis failed for ${fileName}:`, err.message);
    return {
      fileName,
      category: 'general',
      visibility: 'org',
      summary: '无法自动分析',
      mentionedDepartments: [],
      mentionedRoles: [],
    };
  }
}

// ========== 阶段2：Sonnet 整合组织架构 ==========

async function synthesizeWithSonnet(fileAnalyses: FileAnalysis[], fileContents: { fileName: string; content: string }[]): Promise<EnterpriseProfile> {
  // 构建摘要输入
  const summaryBlock = fileAnalyses.map(f =>
    `【${f.fileName}】类型:${f.category} | 摘要:${f.summary} | 提到的部门:${f.mentionedDepartments.join(',')} | 提到的岗位:${f.mentionedRoles.join(',')}`
  ).join('\n');

  // 如果总内容不太长，也把关键文件的部分内容发过去（帮助AI更准确）
  const keyContents = fileContents
    .filter(f => {
      const analysis = fileAnalyses.find(a => a.fileName === f.fileName);
      return analysis && ['jd', 'policy'].includes(analysis.category);
    })
    .slice(0, 5) // 最多5个关键文件
    .map(f => `### ${f.fileName}\n${f.content.slice(0, 3000)}`)
    .join('\n\n');

  try {
    const response = await complexClient.chat.completions.create({
      model: 'claude-sonnet-4-6',
      max_tokens: 4000,
      temperature: 0.2,
      messages: [
        {
          role: 'system',
          content: `你是一个企业组织架构分析专家。根据多份企业文件的分析摘要，整合出完整的企业组织信息。

返回纯 JSON（不要 markdown 代码块）：
{
  "companyName": "从文件中识别的公司全称（找不到就填空字符串）",
  "companyDescription": "公司简介1-2句话（找不到就填空）",
  "departments": [
    {
      "name": "部门名称",
      "description": "部门职责简述（1句话）",
      "children": [
        { "name": "子部门名", "description": "简述" }
      ]
    }
  ],
  "jobRoles": [
    {
      "title": "岗位名称",
      "departmentName": "所属部门（和departments中的name对应）",
      "responsibilities": "核心职责，用分号分隔",
      "boundaries": "不负责的事项（没有就填空）",
      "requiredSkills": "技能要求（没有就填空）"
    }
  ]
}

规则：
- 只提取文件中明确提到的信息，不编造
- 去重：多个文件提到同一个部门只列一次
- 识别层级关系（如"大客户组"隶属于"销售部"放在children里）
- 如果找不到某项信息，对应字段填空字符串或空数组
- departments 和 jobRoles 数组如果完全没有信息就返回空数组`
        },
        {
          role: 'user',
          content: `以下是对一家企业${fileAnalyses.length}份文件的分析结果：

## 文件摘要
${summaryBlock}

## 关键文件内容节选
${keyContents || '（无关键文件内容）'}

请从以上信息中整合出这家企业的组织架构。`
        }
      ],
    });

    const text = response.choices[0]?.message?.content || '';
    const cleaned = text.replace(/```json\s*/g, '').replace(/```\s*/g, '').trim();
    const parsed = JSON.parse(cleaned);

    return {
      companyName: parsed.companyName || '',
      companyDescription: parsed.companyDescription || '',
      departments: parsed.departments || [],
      jobRoles: parsed.jobRoles || [],
      fileClassifications: fileAnalyses,
    };
  } catch (err: any) {
    console.error('[Setup AI] Sonnet synthesis failed:', err.message);
    // 降级：只返回 Haiku 的结果
    return {
      companyName: '',
      companyDescription: '',
      departments: [],
      jobRoles: [],
      fileClassifications: fileAnalyses,
    };
  }

// ========== 主入口：两阶段提取 ==========

export async function extractEnterpriseProfile(files: { fileName: string; content: string }[]): Promise<EnterpriseProfile> {
  console.log(`[Setup AI] Starting extraction for ${files.length} files`);

  // 阶段1：Haiku 逐文件分析（并行，提速）
  console.log('[Setup AI] Phase 1: Haiku file analysis...');
  const fileAnalyses = await Promise.all(
    files.map(f => analyzeFileWithHaiku(f.fileName, f.content))
  );
  console.log(`[Setup AI] Phase 1 complete: ${fileAnalyses.length} files analyzed`);

  // 阶段2：Sonnet 整合
  console.log('[Setup AI] Phase 2: Sonnet synthesis...');
  const profile = await synthesizeWithSonnet(fileAnalyses, files);
  console.log(`[Setup AI] Phase 2 complete: ${profile.departments.length} depts, ${profile.jobRoles.length} roles`);

  return profile;
}
```

## STEP 5：创建 `server/services/setup/setupService.ts`

```typescript
import fs from 'fs';
import path from 'path';
import { storage } from '../../storage';
import { processZipFile, processMultipleFiles, cleanupTempFiles, ExtractedFile } from './zipProcessor';
import { extractEnterpriseProfile, EnterpriseProfile } from './aiExtractor';
import { processDocument } from '../kb/processDocument';

/**
 * 第一步：上传分析
 * 用户上传文件 → 解压 → AI提取 → 返回 EnterpriseProfile 供确认
 */
export async function analyzeUpload(params: {
  zipPath?: string;
  files?: { originalName: string; tempPath: string }[];
}): Promise<{
  profile: EnterpriseProfile;
  extractedFiles: ExtractedFile[];
}> {
  let extractedFiles: ExtractedFile[];

  if (params.zipPath) {
    extractedFiles = await processZipFile(params.zipPath);
  } else if (params.files) {
    extractedFiles = await processMultipleFiles(params.files);
  } else {
    throw new Error('请上传ZIP文件或多个文件');
  }

  if (extractedFiles.length === 0) {
    throw new Error('没有从上传的文件中提取到有效内容。请确保文件格式为 PDF/DOCX/TXT/MD。');
  }

  // AI 提取企业信息
  const profile = await extractEnterpriseProfile(
    extractedFiles.map(f => ({ fileName: f.fileName, content: f.content }))
  );

  return { profile, extractedFiles };
}

/**
 * 第二步：确认写入
 * 用户确认（或修改后确认）AI 提取的结果 → 批量写入数据库
 */
export async function confirmAndSetup(params: {
  orgId: number;
  userId: number;
  profile: EnterpriseProfile;        // 用户确认后的（可能被修改过的）profile
  extractedFiles: ExtractedFile[];    // 原始解压的文件列表
}): Promise<{
  updatedOrg: boolean;
  departmentsCreated: number;
  jobRolesCreated: number;
  documentsCreated: number;
}> {
  const { orgId, userId, profile, extractedFiles } = params;
  let departmentsCreated = 0;
  let jobRolesCreated = 0;
  let documentsCreated = 0;
  let updatedOrg = false;

  // 1. 更新组织名称（如果AI识别到了）
  if (profile.companyName) {
    try {
      await storage.updateOrganization(orgId, {
        name: profile.companyName,
        description: profile.companyDescription || undefined,
      });
      updatedOrg = true;
      console.log(`[Setup] Updated org name: ${profile.companyName}`);
    } catch (err: any) {
      console.warn(`[Setup] Failed to update org:`, err.message);
    }
  }

  // 2. 创建部门（带层级）
  const deptNameToId: Record<string, number> = {};

  for (const dept of profile.departments) {
    try {
      const created = await storage.createDepartment({
        orgId,
        name: dept.name,
        description: dept.description || null,
      });
      deptNameToId[dept.name] = created.id;
      departmentsCreated++;

      // 创建子部门
      if (dept.children && dept.children.length > 0) {
        for (const child of dept.children) {
          try {
            const childCreated = await storage.createDepartment({
              orgId,
              name: child.name,
              description: child.description || null,
              parentDeptId: created.id,
            });
            deptNameToId[child.name] = childCreated.id;
            departmentsCreated++;
          } catch (err: any) {
            console.warn(`[Setup] Failed to create child dept ${child.name}:`, err.message);
          }
        }
      }
    } catch (err: any) {
      console.warn(`[Setup] Failed to create dept ${dept.name}:`, err.message);
    }
  }

  console.log(`[Setup] Created ${departmentsCreated} departments`);

  // 3. 创建岗位角色
  for (const role of profile.jobRoles) {
    try {
      const deptId = deptNameToId[role.departmentName] || null;
      await storage.createJobRole({
        orgId,
        deptId,
        title: role.title,
        responsibilities: role.responsibilities || '',
        boundaries: role.boundaries || null,
        requiredSkills: role.requiredSkills || null,
        description: null,
      });
      jobRolesCreated++;
    } catch (err: any) {
      console.warn(`[Setup] Failed to create role ${role.title}:`, err.message);
    }
  }

  console.log(`[Setup] Created ${jobRolesCreated} job roles`);

  // 4. 把所有文件存入知识库
  for (const file of extractedFiles) {
    try {
      // 找到AI对这个文件的分类
      const classification = profile.fileClassifications.find(
        fc => fc.fileName === file.fileName
      );

      // 移动文件到 kb 目录
      const kbDir = 'uploads/kb/';
      if (!fs.existsSync(kbDir)) fs.mkdirSync(kbDir, { recursive: true });
      
      const newFileName = `${Date.now()}_${Math.round(Math.random() * 1e9)}_${file.fileName}`;
      const kbPath = path.join(kbDir, newFileName);
      
      // 复制文件到 KB 目录（不是移动，因为临时文件后面统一清理）
      fs.copyFileSync(file.filePath, kbPath);

      // 处理 visibleDeptIds
      let visibleDeptIds: string | null = null;
      if (classification?.visibility === 'department' && classification?.visibleDepartment) {
        const deptId = deptNameToId[classification.visibleDepartment];
        if (deptId) {
          visibleDeptIds = JSON.stringify([deptId]);
        }
      }

      // 创建知识库文档记录
      const doc = await storage.createKbDocument({
        orgId,
        uploadedBy: userId,
        title: file.fileName.replace(/\.[^/.]+$/, ''), // 去掉扩展名
        fileName: file.fileName,
        fileType: file.fileType,
        fileSize: file.fileSize,
        fileUrl: `/uploads/kb/${newFileName}`,
        category: classification?.category || 'general',
        visibility: classification?.visibility || 'org',
        visibleDeptIds,
        status: 'pending',
        chunkCount: 0,
      });

      documentsCreated++;

      // 异步触发文档处理（切片 + embedding）
      setImmediate(() => {
        processDocument(doc.id).catch(err => {
          console.error(`[Setup] KB processing failed for ${file.fileName}:`, err);
        });
      });

    } catch (err: any) {
      console.warn(`[Setup] Failed to create KB doc ${file.fileName}:`, err.message);
    }
  }

  console.log(`[Setup] Created ${documentsCreated} KB documents`);

  // 5. 清理临时文件
  cleanupTempFiles(extractedFiles);

  // 6. 记录操作日志
  await storage.createActivityLog({
    orgId,
    userId,
    entityType: 'organization',
    entityId: orgId,
    action: 'smart_setup',
    changes: JSON.stringify({
      departmentsCreated,
      jobRolesCreated,
      documentsCreated,
      companyName: profile.companyName,
    }),
    source: 'ai',
  });

  return { updatedOrg, departmentsCreated, jobRolesCreated, documentsCreated };
}
```

**注意**：`storage.createDepartment` 和 `storage.createJobRole` 需要检查实际参数名。查看 `server/storage.ts` 中这两个方法的签名，确保传参匹配。特别是：

- createDepartment 的参数可能需要 `orgId` 字段名匹配（是 `orgId` 还是 `org_id`）
- createJobRole 同理
- 如果 storage 中没有 `createJobRole` 方法，需要新增

```typescript
    const userRole = req.userRole || 'member';
    if (userRole !== 'owner') {
      return res.status(403).json({ error: '仅组织所有者可执行初始化设置' });
    }

    const uploadedFiles = req.files as Express.Multer.File[];
    if (!uploadedFiles || uploadedFiles.length === 0) {
      return res.status(400).json({ error: '请上传文件' });
    }

    // 判断是 ZIP 还是多文件
    const zipFile = uploadedFiles.find(f => f.originalname.toLowerCase().endsWith('.zip'));

    let result;
    if (zipFile) {
      // ZIP 模式
      result = await analyzeUpload({ zipPath: zipFile.path });
      // 清理 ZIP 临时文件
      if (fs.existsSync(zipFile.path)) fs.unlinkSync(zipFile.path);
    } else {
      // 多文件模式
      result = await analyzeUpload({
        files: uploadedFiles.map(f => ({
          originalName: f.originalname,
          tempPath: f.path,
        })),
      });
    }

    // 把 extractedFiles 的 filePath 信息缓存起来（用于确认后写入 KB）
    // 存在内存中，用 sessionId 或 orgId 做 key
    // 简单方案：存在临时 JSON 文件中
    const cacheKey = `setup_cache_${req.orgId}_${Date.now()}`;
    const cachePath = `uploads/${cacheKey}.json`;
    fs.writeFileSync(cachePath, JSON.stringify({
      extractedFiles: result.extractedFiles.map(f => ({
        fileName: f.fileName,
        filePath: f.filePath,
        fileType: f.fileType,
        fileSize: f.fileSize,
      })),
    }));

    return res.json({
      data: {
        profile: result.profile,
        cacheKey,  // 前端确认时需要传回来
        fileCount: result.extractedFiles.length,
      }
    });
  } catch (e: any) {
    console.error('[Setup] Analyze error:', e);
    return res.status(500).json({ error: e.message });
  }
});

/**
 * 确认 AI 分析结果，批量写入数据库
 */
app.post("/api/setup/confirm", authMiddleware, async (req: any, res) => {
  try {
    const userRole = req.userRole || 'member';
    if (userRole !== 'owner') {
      return res.status(403).json({ error: '仅组织所有者可执行此操作' });
    }

    const { profile, cacheKey } = req.body;
    if (!profile || !cacheKey) {
      return res.status(400).json({ error: '缺少 profile 或 cacheKey' });
    }

    // 读取缓存的文件信息
    const cachePath = `uploads/${cacheKey}.json`;
    if (!fs.existsSync(cachePath)) {
      return res.status(400).json({ error: '分析结果已过期，请重新上传' });
    }
    const cached = JSON.parse(fs.readFileSync(cachePath, 'utf-8'));

    // 执行批量写入
    const result = await confirmAndSetup({
      orgId: req.orgId,
      userId: req.currentUserId,
      profile,
      extractedFiles: cached.extractedFiles,
    });

    // 清理缓存
    if (fs.existsSync(cachePath)) fs.unlinkSync(cachePath);

    return res.json({ data: result });
  } catch (e: any) {
    console.error('[Setup] Confirm error:', e);
    return res.status(500).json({ error: e.message });
  }
});

/**
 * 获取当前组织的初始化状态（已有多少部门、岗位、文档）
 */
app.get("/api/setup/status", authMiddleware, async (req: any, res) => {
  try {
    const departments = await storage.getDepartmentsByOrg(req.orgId);
    const jobRoles = await storage.getJobRolesByOrg(req.orgId);
    const kbDocs = await storage.getKbDocumentsByOrg(req.orgId);
    const members = await storage.getOrgMembers(req.orgId);

    return res.json({
      data: {
        departmentCount: departments?.length || 0,
        jobRoleCount: jobRoles?.length || 0,
        kbDocumentCount: kbDocs?.length || 0,
        memberCount: members?.length || 0,
        isSetupComplete: (departments?.length || 0) > 0 || (kbDocs?.length || 0) > 0,
      }
    });
  } catch (e: any) {
    return res.status(500).json({ error: e.message });
  }
});
```

**注意关于缓存**：`/api/setup/analyze` 返回的文件路径需要在 `/api/setup/confirm` 中使用。这里用了临时 JSON 文件缓存。更好的方案是用 Redis 或数据库缓存，但对MVP来说文件缓存够用。

**注意关于 storage 方法**：路由中用到了几个方法，需要确认 storage.ts 中是否有：

- `storage.getDepartmentsByOrg(orgId)` — 可能叫 `getDepartments()` 并已按orgId过滤
- `storage.getJobRolesByOrg(orgId)` — 可能叫 `getJobRoles()`
- `storage.getOrgMembers(orgId)` — 可能叫 `getOrgMembers()` 或 `getMembersByOrg()`
- `storage.createJobRole(data)` — 确认参数结构

让 Replit Agent 检查 storage.ts，用实际存在的方法名替换。
